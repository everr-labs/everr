everr local query 'SELECT toString(LogAttributes.everr.recording.clip.id) AS clip, toString(LogAttributes.everr.recording.segment.index) AS segment, toString(LogAttributes.everr.recording.event.index) AS event_index, toString(LogAttributes.everr.recording.event.piece) AS piece, toString(LogAttributes.everr.recording.event.pieces) AS pieces, toString(LogAttributes.everr.recording.stream.id) AS stream, toString(LogAttributes.session.id) AS session, Timestamp, Body FROM logs WHERE EventName = 'everr.browser.recording.segment' AND ServiceName = 'verify-recording-web-v4' ORDER BY clip, toUInt32OrZero(segment)' --format ndjson
